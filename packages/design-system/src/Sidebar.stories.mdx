export const Default = () => {
  const [selectedVersion, setSelectedVersion] = useState(sidebarData.versions[0])

return (

<SidebarProvider>
  <Sidebar>
    <SidebarHeader>
      <SidebarMenu>
        <SidebarMenuItem>
          <DropdownMenu>
            <DropdownMenuTrigger asChild>
              <SidebarMenuButton
                size="lg"
                className="data-[state=open]:bg-sidebar-accent data-[state=open]:text-sidebar-accent-foreground"
              >
                <div className="bg-sidebar-primary text-sidebar-primary-foreground flex aspect-square size-8 items-center justify-center rounded-lg">
                  <GalleryVerticalEnd className="size-4" />
                </div>
                <div className="flex flex-col gap-0.5 leading-none">
                  <span className="font-medium">Documentation</span>
                  <span className="">v{selectedVersion}</span>
                </div>
                <ChevronsUpDown className="ml-auto" />
              </SidebarMenuButton>
            </DropdownMenuTrigger>
            <DropdownMenuContent
              className="w-(--radix-dropdown-menu-trigger-width)"
              align="start"
            >
              {sidebarData.versions.map((version) => (
                <DropdownMenuItem
                  key={version}
                  onSelect={() => setSelectedVersion(version)}
                >
                  v{version}{' '}
                  {version === selectedVersion && <Check className="ml-auto" />}
                </DropdownMenuItem>
              ))}
            </DropdownMenuContent>
          </DropdownMenu>
        </SidebarMenuItem>
      </SidebarMenu>
      <form>
        <SidebarGroup className="py-0">
          <SidebarGroupContent className="relative">
            <Label htmlFor="search" className="sr-only">
              Search
            </Label>
            <SidebarInput
              id="search"
              placeholder="Search the docs..."
              className="pl-8"
            />
            <Search className="pointer-events-none absolute top-1/2 left-2 size-4 -translate-y-1/2 opacity-50 select-none" />
          </SidebarGroupContent>
        </SidebarGroup>
      </form>
    </SidebarHeader>
    <SidebarContent>
      {sidebarData.navMain.map((item) => (
        <SidebarGroup key={item.title}>
          <SidebarGroupLabel>{item.title}</SidebarGroupLabel>
          <SidebarGroupContent>
            <SidebarMenu>
              {item.items.map((item) => (
                <SidebarMenuItem key={item.title}>
                  <SidebarMenuButton asChild isActive={item.isActive}>
                    <a href={item.url}>{item.title}</a>
                  </SidebarMenuButton>
                </SidebarMenuItem>
              ))}
            </SidebarMenu>
          </SidebarGroupContent>
        </SidebarGroup>
      ))}
    </SidebarContent>
    <SidebarRail />
  </Sidebar>
  <SidebarInset>
    <header className="flex h-16 shrink-0 items-center gap-2 border-b px-4">

      />

          <BreadcrumbItem className="hidden md:block">

          <BreadcrumbSeparator className="hidden md:block" />

export const Compatibility = () => {
  const contentRef = useRef(null)
  const [side, setSide] = useState('left')
  const [direction, setDirection] = useState('ltr')
  const [collapsible, setCollapsible] = useState('offcanvas')
  const [desktopOpen, setDesktopOpen] = useState(true)
  const [desktopChanges, setDesktopChanges] = useState(0)
  const [refState, setRefState] = useState('null')

  return (
    <div
      className="bg-background text-foreground min-h-svh overflow-x-hidden"
      data-test="sidebar-compatibility"
    >
      <div className="flex flex-wrap items-center gap-2 border-b p-2 pt-16 text-xs">
        <button
          type="button"
          className="rounded border px-2 py-1"
          data-test="sidebar-side-left"
          onClick={() => setSide('left')}
        >
          Side left
        </button>
        <button
          type="button"
          className="rounded border px-2 py-1"
          data-test="sidebar-side-right"
          onClick={() => setSide('right')}
        >
          Side right
        </button>
        <button
          type="button"
          className="rounded border px-2 py-1"
          data-test="sidebar-collapsible-offcanvas"
          onClick={() => setCollapsible('offcanvas')}
        >
          Offcanvas
        </button>
        <button
          type="button"
          className="rounded border px-2 py-1"
          data-test="sidebar-collapsible-icon"
          onClick={() => setCollapsible('icon')}
        >
          Icon
        </button>
        <button
          type="button"
          className="rounded border px-2 py-1"
          data-test="sidebar-dir-ltr"
          onClick={() => setDirection('ltr')}
        >
          Direction ltr
        </button>
        <button
          type="button"
          className="rounded border px-2 py-1"
          data-test="sidebar-dir-rtl"
          onClick={() => setDirection('rtl')}
        >
          Direction rtl
        </button>
        <output data-test="sidebar-side-state">{side}</output>
        <output data-test="sidebar-direction-state">{direction}</output>
        <output data-test="sidebar-collapsible-state">{collapsible}</output>
        <output data-test="sidebar-controlled-state">
          {String(desktopOpen)}
        </output>
        <output data-test="sidebar-desktop-changes">{desktopChanges}</output>
      </div>

      <SidebarProvider
        open={desktopOpen}
        onOpenChange={(open) => {
          setDesktopOpen(open)
          setDesktopChanges((count) => count + 1)
        }}
      >
        <Sidebar
          ref={contentRef}
          id="sidebar-compatibility-content"
          data-test="sidebar-shell"
          data-caller="sidebar-compatibility"
          aria-roledescription="Application navigation"
          className="caller-sidebar-class"
          style={{
            outline: '2px solid transparent',
            outlineOffset: '3px',
          }}
          dir={direction}
          side={side}
          collapsible={collapsible}
        >
          <SidebarHeader className="p-3">
            <p className="text-sm font-semibold">Compatibility navigation</p>
            <button
              type="button"
              className="w-fit rounded border px-2 py-1 text-xs"
              data-test="sidebar-read-ref"
              onClick={() => {
                const element = contentRef.current
                setRefState(
                  element
                    ? JSON.stringify({
                        id: element.id,
                        slot: element.getAttribute('data-slot'),
                        caller: element.getAttribute('data-caller'),
                        dir: element.getAttribute('dir'),
                      })
                    : 'null'
                )
              }}
            >
              Read content ref
            </button>
            <output data-test="sidebar-ref-state">{refState}</output>
          </SidebarHeader>
          <SidebarContent>
            <SidebarGroup>
              <SidebarGroupLabel>Navigation</SidebarGroupLabel>
              <SidebarGroupContent>
                <SidebarMenu>
                  <SidebarMenuItem>
                    <SidebarMenuButton asChild>
                      <a href="#sidebar-overview">Overview</a>
                    </SidebarMenuButton>
                  </SidebarMenuItem>
                  <SidebarMenuItem>
                    <SidebarMenuButton asChild>
                      <a href="#sidebar-settings">Settings</a>
                    </SidebarMenuButton>
                  </SidebarMenuItem>
                </SidebarMenu>
              </SidebarGroupContent>
            </SidebarGroup>
          </SidebarContent>
        </Sidebar>

        <SidebarInset className="min-w-0">
          <header className="flex min-h-16 flex-wrap items-center gap-2 border-b p-2">
            <SidebarTrigger
              id="sidebar-story-trigger-a"
              data-test="sidebar-trigger-a"
            />
            <SidebarTrigger
              id="sidebar-story-trigger-b"
              data-test="sidebar-trigger-b"
            />
            <button
              id="sidebar-story-shortcut-opener"
              type="button"
              className="rounded border px-2 py-1 text-xs"
              data-test="sidebar-shortcut-opener"
            >
              Shortcut surface
            </button>
          </header>
          <div className="min-w-0 p-3">
            <div
              className="w-60 max-w-full overflow-hidden rounded border"
              data-test="sidebar-static-shell"
            >
              <Sidebar
                id="sidebar-story-static"
                data-test="sidebar-static"
                role="navigation"
                aria-label="Static navigation"
                collapsible="none"
                className="h-24"
                dir={direction}
              >
                <SidebarHeader className="p-2 text-xs">
                  Static sidebar
                </SidebarHeader>
                <SidebarContent>
                  <SidebarMenu>
                    <SidebarMenuItem>
                      <SidebarMenuButton asChild>
                        <a href="#sidebar-static-link">Static link</a>
                      </SidebarMenuButton>
                    </SidebarMenuItem>
                  </SidebarMenu>
                </SidebarContent>
              </Sidebar>
            </div>
          </div>
        </SidebarInset>
      </SidebarProvider>
    </div>
  )
}
