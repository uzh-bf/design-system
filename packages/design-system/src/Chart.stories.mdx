import { Bar, BarChart, CartesianGrid, XAxis } from 'recharts'
import {
  ChartContainer,
  ChartLegend,
  ChartLegendContent,
  ChartTooltip,
  ChartTooltipContent,
} from './Chart'

{/* START README */}
<div className="prose prose-sm max-w-none">
The `Chart` exports are the design-system's [Recharts](https://recharts.org)
wrapper (the shadcn chart primitives). `ChartContainer` provides a responsive,
token-themed canvas; `ChartTooltip` / `ChartTooltipContent` and `ChartLegend` /
`ChartLegendContent` give consistent tooltips and legends.

You drive the colors through a `config` object keyed by data series. Each key
maps to a `label` and a `color`; the container emits CSS variables
(`--color-<key>`) you reference from the Recharts marks (e.g.
`fill="var(--color-desktop)"`). Point a series `color` at a DS token (such as
`var(--color-primary-100)`) and the chart recolors with the active theme.
</div>

Technical Overview: Recharts wrapper. ChartContainer takes a `config` (series → label+color), emits `--color-<key>` CSS vars, and renders a responsive container. ChartTooltipContent / ChartLegendContent are themed tooltip/legend renderers.
Dependencies: recharts

Available Stories:

- Default: A two-series bar chart with tooltip and legend

Exports:

- ChartContainer - Responsive themed canvas; requires a `config` prop
- ChartTooltip / ChartTooltipContent - Recharts tooltip + themed content
- ChartLegend / ChartLegendContent - Recharts legend + themed content
- ChartStyle - Injects the per-series CSS color variables

Usage Examples:

```tsx

  desktop: { label: 'Desktop', color: 'var(--color-primary-100)' },
}

<ChartContainer config={config}>

    <Bar dataKey="desktop" fill="var(--color-desktop)" />
  </BarChart>
</ChartContainer>
```

export const Default = () => {
  const chartData = [
    { month: 'January', desktop: 186, mobile: 80 },
    { month: 'February', desktop: 305, mobile: 200 },
    { month: 'March', desktop: 237, mobile: 120 },
    { month: 'April', desktop: 73, mobile: 190 },
    { month: 'May', desktop: 209, mobile: 130 },
    { month: 'June', desktop: 214, mobile: 140 },
  ]
  const chartConfig = {
    desktop: { label: 'Desktop', color: 'var(--color-primary-100)' },
    mobile: { label: 'Mobile', color: 'var(--color-primary-40)' },
  }
  return (
    <ChartContainer config={chartConfig} className="h-[300px] w-[600px]">
      <BarChart data={chartData}>
        <CartesianGrid vertical={false} />
        <XAxis
          dataKey="month"
          tickLine={false}
          axisLine={false}
          tickMargin={10}
          tickFormatter={(value) => value.slice(0, 3)}
        />
        <ChartTooltip content={<ChartTooltipContent />} />
        <ChartLegend content={<ChartLegendContent />} />
        <Bar
          dataKey="desktop"
          fill="var(--color-desktop)"
          radius={4}
          isAnimationActive={false}
        />
        <Bar
          dataKey="mobile"
          fill="var(--color-mobile)"
          radius={4}
          isAnimationActive={false}
        />
      </BarChart>
    </ChartContainer>
  )
}
