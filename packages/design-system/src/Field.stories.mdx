import { Input } from './ui/input'
import {
  Field,
  FieldDescription,
  FieldError,
  FieldGroup,
  FieldLabel,
} from './Field'

{/* START README */}
<div className="prose prose-sm max-w-none">
The `Field` component is a layout primitive for form rows: it aligns a label, a
control, an optional description, and validation errors with consistent spacing
and orientation. It is presentation-only and does not manage form state.

Use this component when you need to:

- Lay out a label + control + helper text + error as one accessible unit
- Switch a field between `vertical`, `horizontal`, and `responsive` orientation
- Group several fields with `FieldGroup`, `FieldSet`, and `FieldLegend`

`Field` does not bind form state. For Formik-managed inputs in this library,
keep using the `Formik*` field components; reach for `Field` when you are
composing custom controls and only need the layout and a11y wiring.
</div>
{/* END README */}

- FieldSeparator: Divider (with optional centered label) between fields

Related Components:

- InputGroup: A control with addons that can sit inside a Field

export const Default = () => {
  return (
    <FieldGroup className="max-w-sm">
      <Field>
        <FieldLabel htmlFor="email">Email</FieldLabel>
        <Input id="email" type="email" placeholder="you@example.com" />
        <FieldDescription>We never share your address.</FieldDescription>
      </Field>
    </FieldGroup>
  )
}

export const WithError = () => {
  return (
    <FieldGroup className="max-w-sm">
      <Field data-invalid="true">
        <FieldLabel htmlFor="name">Name</FieldLabel>
        <Input
          id="name"
          aria-invalid="true"
          className="border-destructive-text!"
        />
        <FieldError errors={[{ message: 'Name is required.' }]} />
      </Field>
    </FieldGroup>
  )
}
