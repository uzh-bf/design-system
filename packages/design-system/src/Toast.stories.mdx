export const SonnerRootTheme = () => {
  useRestoreDocumentRoot()
  const [rootDark, setRootDark] = useState(false)

  useEffect(() => {
    setRootDark(document.documentElement.classList.contains('dark'))
  }, [])

  const toggleRootDark = () => {
    const root = document.documentElement
    const next = !root.classList.contains('dark')
    root.classList.toggle('dark', next)
    setRootDark(next)
  }

  return (
    <>
      <Toaster
        id="sonner-root-theme-toaster"
        closeButton
        toastOptions={{ duration: 60000 }}
      />
      <div className="bg-background text-foreground flex max-w-full flex-wrap items-center gap-2 p-4 pt-28">
        <button
          type="button"
          data-test="trigger-root-theme-toast"
          className="rounded border px-3 py-2 text-sm"
          onClick={() =>
            toast({
              message: 'Root theme compatibility toast',
              options: {
                duration: 60000,
                description: 'This toast stays mounted across root changes.',
                testId: 'sonner-root-theme-toast',
                toasterId: 'sonner-root-theme-toaster',
              },
            })
          }
        >
          Show root theme toast
        </button>
        <button
          type="button"
          data-test="toggle-root-dark"
          className="rounded border px-3 py-2 text-sm"
          onClick={toggleRootDark}
        >
          Toggle root dark mode
        </button>
        <output data-test="root-dark">{String(rootDark)}</output>
      </div>
    </>
  )
}

export const SonnerProviderTheme = () => {
  useRestoreDocumentRoot()
  const [explicitDark, setExplicitDark] = useState(false)

  useEffect(() => {
    document.documentElement.classList.add('dark')
  }, [])

  return (
    <NextThemeProvider
      attribute="data-next-mode"
      defaultTheme="dark"
      forcedTheme="light"
      enableSystem={false}
      enableColorScheme={false}
    >
      <Toaster
        id="sonner-provider-theme-toaster"
        theme={explicitDark ? 'dark' : undefined}
        toastOptions={{ duration: 60000 }}
      />
      <div className="bg-background text-foreground flex max-w-full flex-wrap gap-2 p-4 pt-28">
        <button
          type="button"
          data-test="trigger-provider-theme-toast"
          className="rounded border px-3 py-2 text-sm"
          onClick={() =>
            toast({
              message: 'Provider theme compatibility toast',
              options: {
                duration: 60000,
                testId: 'sonner-provider-theme-toast',
                toasterId: 'sonner-provider-theme-toaster',
              },
            })
          }
        >
          Show provider theme toast
        </button>
        <button
          type="button"
          data-test="toggle-explicit-dark"
          className="rounded border px-3 py-2 text-sm"
          onClick={() => setExplicitDark((enabled) => !enabled)}
        >
          Toggle explicit dark theme
        </button>
      </div>
    </NextThemeProvider>
  )
}
