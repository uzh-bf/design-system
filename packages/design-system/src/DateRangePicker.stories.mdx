import { useState } from 'react'
import { DateRangePicker } from './DateRangePicker'

{/* START README */}
<div className="prose prose-sm max-w-none">
The `DateRangePicker` selects a date range (from–to). It composes the `Popover`
and `Calendar` primitives, driving the calendar in `range` mode with two months
shown side by side. The open state is internal; the selected range is controlled
via `range` / `onRangeChange`.

The `range` value is a react-day-picker `DateRange` (`{ from?: Date; to?: Date }`).
Pick a start day, then an end day; the trigger shows `DD.MM.YYYY – DD.MM.YYYY`
(and `… ` while only the start is chosen).
</div>

export const Default = () => {
  const [range, setRange] = useState(undefined)
  return (
    <DateRangePicker
      range={range}
      onRangeChange={setRange}
      placeholder="Pick a date range…"
    />
  )
}

export const Preselected = () => {
  // June 8–15, 2026 (JS Date month is 0-indexed, so 5 = June)
  const [range, setRange] = useState({
    from: new Date(2026, 5, 8),
    to: new Date(2026, 5, 15),
  })
  return <DateRangePicker range={range} onRangeChange={setRange} />
}

export const Disabled = () => {
  const [range, setRange] = useState(undefined)
  return (
    <DateRangePicker
      range={range}
      onRangeChange={setRange}
      disabled
      placeholder="Disabled"
    />
  )
}
