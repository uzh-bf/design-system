{/* START README */}
<div className="prose prose-sm max-w-none">
The `Combobox` is a searchable single-select. It composes the `Popover` and
`Command` primitives into one component: click the trigger to open a filterable
list, type to narrow it, and pick an item. The open state is internal; the
selected `value` is controlled externally via `value` / `onChange`.

Use this component when a plain `Select` is unwieldy — long option lists, or when
users benefit from type-ahead filtering.

Each item needs a unique string `value` and a `label`. String labels are passed
to cmdk as explicit search keywords; non-string (ReactNode) labels are still
filterable via their rendered text.
</div>

export const Default = () => {
  const [value, setValue] = useState('')
  const frameworks = [
    { value: 'next', label: 'Next.js' },
    { value: 'sveltekit', label: 'SvelteKit' },
    { value: 'nuxt', label: 'Nuxt.js' },
    { value: 'remix', label: 'Remix' },
    { value: 'astro', label: 'Astro' },
    { value: 'vite', label: 'Vite' },
  ]
  return (
    <Combobox
      items={frameworks}
      value={value}
      onChange={setValue}
      placeholder="Select framework…"
      searchPlaceholder="Search framework…"
    />
  )
}

export const Preselected = () => {
  const [value, setValue] = useState('vite')
  const frameworks = [
    { value: 'next', label: 'Next.js' },
    { value: 'remix', label: 'Remix' },
    { value: 'vite', label: 'Vite' },
  ]
  return <Combobox items={frameworks} value={value} onChange={setValue} />
}

export const Disabled = () => {
  const [value, setValue] = useState('')
  return (
    <Combobox
      items={[{ value: 'vite', label: 'Vite' }]}
      value={value}
      onChange={setValue}
      disabled
      placeholder="Disabled"
    />
  )
}
